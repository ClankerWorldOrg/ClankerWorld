#!/bin/bash
set -euo pipefail
export PATH=/workspace/tools/dotnet:$PATH DOTNET_PROCESSOR_COUNT=4 ClankerWorld__Clock__StrictTickFailures=true DOTNET_CLI_USE_MSBUILD_SERVER=0 MSBUILDDISABLENODEREUSE=1
cd /workspace/cw-2550-membership
r=.evidence/keep/2550/membership
dotnet restore --locked-mode > "$r/restore.log" 2>&1
dotnet build -c Release --no-restore > "$r/build.log" 2>&1
dotnet test tests/ClankerWorld.Tests/ClankerWorld.Tests.csproj -c Release --no-build --filter 'FullyQualifiedName~ClankerWorld.Tests.TownMembershipTests|FullyQualifiedName~ClankerWorld.Tests.GuardianPlacementTests|FullyQualifiedName~ClankerWorld.Tests.GuardianPlacementBoundaryTests|FullyQualifiedName~ClankerWorld.Tests.HouseholdJoinRequestTests|FullyQualifiedName~ClankerWorld.Tests.RuntimeStateBoundaryTests' --results-directory "$r/focused" --logger 'trx;LogFileName=tests.trx' --logger 'console;verbosity=normal' -- xUnit.MaxParallelThreads=4 > "$r/focused.log" 2>&1
python skills/test-audit/scripts/trx_summary.py "$r/focused" > "$r/focused-summary.json"
bash scripts/compare-tick-equivalence.sh /workspace/cw-2550-main-ed54 /workspace/cw-2550-membership "$PWD/$r/native-equivalence" > "$r/native-equivalence.log" 2>&1
dotnet format --verify-no-changes --no-restore > "$r/format.log" 2>&1
git diff --check
python - <<'PY'
from pathlib import Path
import json
Path('.evidence/keep/2550/membership/local-result.json').write_text(json.dumps({'locked_restore':'passed','Release_build':'passed','focused':'passed seefocused-summary','native_equivalence':'68framespassed seelog','whole_format':'passed','diff_check':'passed'})+'\n');print('All membership local checks passed',flush=True)
PY
