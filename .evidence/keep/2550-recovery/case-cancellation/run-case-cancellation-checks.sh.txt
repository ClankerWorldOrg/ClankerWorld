#!/bin/bash
set -euo pipefail
export PATH=/workspace/tools/dotnet:$PATH DOTNET_PROCESSOR_COUNT=4 ClankerWorld__Clock__StrictTickFailures=true DOTNET_CLI_USE_MSBUILD_SERVER=0 MSBUILDDISABLENODEREUSE=1
cd /workspace/cw-2550-case-cancellation
r=.evidence/keep/2550/case-cancellation
dotnet restore --locked-mode > "$r/restore.log" 2>&1
dotnet build -c Release --no-restore > "$r/build.log" 2>&1
dotnet test tests/ClankerWorld.Tests/ClankerWorld.Tests.csproj -c Release --no-build --filter 'FullyQualifiedName~ClankerWorld.Tests.TownHearingProcedureTests|FullyQualifiedName~ClankerWorld.Tests.TownLandHearingRulesTests|FullyQualifiedName~ClankerWorld.Tests.TownNonLandRulesTests|FullyQualifiedName~ClankerWorld.Tests.TownLandHearingRuntimeTests|FullyQualifiedName~ClankerWorld.Tests.TownLandCasePartySuccessionTests|FullyQualifiedName~ClankerWorld.Tests.NonLandJudgePlanTests|FullyQualifiedName~ClankerWorld.Tests.NonLandJudgeHeldBallotTests|FullyQualifiedName~ClankerWorld.Tests.TownNonLandAuthorityRuntimeTests' --results-directory "$r/focused" --logger 'trx;LogFileName=tests.trx' --logger 'console;verbosity=normal' -- xUnit.MaxParallelThreads=4 > "$r/focused.log" 2>&1
python skills/test-audit/scripts/trx_summary.py "$r/focused" > "$r/focused-summary.json"
bash scripts/compare-tick-equivalence.sh /workspace/cw-2550-main-91f4 /workspace/cw-2550-case-cancellation "$PWD/$r/native-equivalence" > "$r/native-equivalence.log" 2>&1
dotnet format --verify-no-changes --no-restore > "$r/format.log" 2>&1
git diff --check
python - <<'PY'
from pathlib import Path
import json
Path('.evidence/keep/2550/case-cancellation/local-result.json').write_text(json.dumps({'locked_restore':'passed','Release_build':'passed','focused':'passed; see focused-summary','native_equivalence':'68 frames passed; see log','whole_format':'passed','diff_check':'passed'})+'\n');print('Case-cancellation local gates passed',flush=True)
PY
